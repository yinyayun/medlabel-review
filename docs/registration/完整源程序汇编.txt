def main():
    parser = argparse.ArgumentParser(description="MedLabel Review 本地说明书审核工作台")
    parser.add_argument("--host", default="127.0.0.1")
    parser.add_argument("--port", type=int, default=8000)

from datetime import date
import hashlib
import json
import os
from pathlib import Path

from urllib.parse import urlsplit
from uuid import uuid4

from fastapi import FastAPI, File, Form, HTTPException, Request, UploadFile

from fastapi.staticfiles import StaticFiles

from app.comparison import compare_sections

from app.demo import DRUG, NEW_TEXT, OLD_TEXT

from app.reports import export_report

MAX_FILE_SIZE = 10 * 1024 * 1024
MAX_TEXT = 300_000
STATIC_DIR = Path(__file__).parent / "static"


class Input(BaseModel):
    model_config = ConfigDict(extra="forbid", str_strip_whitespace=True)


class DrugInput(Input):
    name: str = Field(min_length=1, max_length=200)
    manufacturer: str = Field(default="", max_length=300)
    specification: str = Field(default="", max_length=300)
    approval_number: str = Field(default="", max_length=200)
    notes: str = Field(default="", max_length=3000)


class TextVersionInput(Input):
    label: str = Field(min_length=1, max_length=120)
    version_date: date
    source: str = Field(default="", max_length=2000)
    text: str = Field(min_length=1, max_length=MAX_TEXT)


class SectionInput(Input):
    title: str = Field(min_length=1, max_length=200)
    text: str = Field(max_length=MAX_TEXT)


class SectionsInput(Input):
    sections: list[SectionInput] = Field(min_length=1, max_length=200)


class ComparisonInput(Input):
    old_version_id: int = Field(gt=0)
    new_version_id: int = Field(gt=0)


class ReviewInput(Input):
    status: Literal["pending", "confirmed", "ignored"]
    note: str = Field(default="", max_length=3000)
    revision: int = Field(ge=0)


def required_row(connection, table, entity_id):
    # Table names are exclusively supplied by internal call sites.
    if table not in {"drugs", "versions", "changes", "comparisons"}:
        raise ValueError("Unknown table")
    row = connection.execute(f"SELECT * FROM {table} WHERE id=?", (entity_id,)).fetchone()
    if row is None:
        raise HTTPException(404, "记录不存在")
    return row


def version_data(row, full=False):
    result = dict(row)
    sections = json.loads(result.pop("sections"))
    result.pop("storage_name", None)
    result["warnings"] = json.loads(result["warnings"])
    result["section_count"] = len(sections)
    if full:
        result["sections"] = sections
    else:
        result.pop("text", None)
    return result


def change_data(row):
    result = json.loads(row["payload"])
    result.update({key: row[key] for key in ("id", "comparison_id", "status", "note", "reviewer", "reviewed_at", "revision")})
    return result


def comparison_data(connection, entity_id, full=True):
    row = connection.execute(
        """SELECT c.*, d.name AS drug_name, ov.label AS old_label, nv.label AS new_label
        FROM comparisons c JOIN drugs d ON d.id=c.drug_id
        JOIN versions ov ON ov.id=c.old_version_id
        JOIN versions nv ON nv.id=c.new_version_id WHERE c.id=?""", (entity_id,)
    ).fetchone()
    if row is None:
        raise HTTPException(404, "比对记录不存在")
    result = dict(row)
    result["warnings"] = json.loads(result["warnings"])
    changes = [change_data(item) for item in connection.execute(
        "SELECT * FROM changes WHERE comparison_id=? ORDER BY id", (entity_id,)
    )]
    result["total_changes"] = len(changes)
    result["pending_changes"] = sum(item["status"] == "pending" for item in changes)
    result["status"] = "pending" if result["pending_changes"] else "completed"
    if full:
        result["changes"] = changes
        result["old_version"] = version_data(required_row(connection, "versions", result["old_version_id"]), True)
        result["new_version"] = version_data(required_row(connection, "versions", result["new_version_id"]), True)
    return result


def add_drug(connection, payload, actor):
    cursor = connection.execute(
        "INSERT INTO drugs(name,manufacturer,specification,approval_number,notes,created_at) VALUES(?,?,?,?,?,?)",
        (payload["name"], payload.get("manufacturer", ""), payload.get("specification", ""),
         payload.get("approval_number", ""), payload.get("notes", ""), now()),
    )
    audit(connection, actor, "创建药品档案", "drug", cursor.lastrowid, {"name": payload["name"]})
    return cursor.lastrowid


def add_version(connection, db, drug_id, label, version_date, source, filename, data, actor, created_files):
    required_row(connection, "drugs", drug_id)
    if not label.strip() or len(label.strip()) > 120:
        raise HTTPException(422, "版本名称须为 1–120 个字符")
    if len(source) > 2000:
        raise HTTPException(422, "来源信息不能超过 2000 个字符")
    if len(data) > MAX_FILE_SIZE:
        raise HTTPException(413, "文件不能超过 10 MB")
    if not data:
        raise HTTPException(422, "文件不能为空")
    try:
        parsed_date = date.fromisoformat(str(version_date)).isoformat()
    except (TypeError, ValueError):
        raise HTTPException(422, "版本日期格式应为 YYYY-MM-DD")
    clean_filename = Path(filename.replace("\\", "/")).name[:200]
    if not clean_filename:
        raise HTTPException(422, "文件名不能为空")
    try:
        text = extract_text(clean_filename, data)
        if len(text) > MAX_TEXT:
            raise ValueError("文档提取后不能超过 300000 个字符，请拆分文档")
        sections = parse_sections(text)
    except ValueError as error:
        raise HTTPException(422, str(error))
    warnings = []
    if len(sections) == 1:
        warnings.append("仅识别到一个章节，请在开始比对前核对章节划分。")
    storage_name = uuid4().hex + Path(clean_filename).suffix.lower()
    path = db.uploads / storage_name
    path.write_bytes(data)
    created_files.append(path)
    digest = hashlib.sha256(data).hexdigest()
    cursor = connection.execute(
        """INSERT INTO versions(drug_id,label,version_date,source,filename,sha256,storage_name,
        text,sections,warnings,created_at) VALUES(?,?,?,?,?,?,?,?,?,?,?)""",
        (drug_id, label.strip(), parsed_date, source.strip(), clean_filename, digest, storage_name,
         text, encode(sections), encode(warnings), now()),
    )
    audit(connection, actor, "导入说明书版本", "version", cursor.lastrowid,
          {"drug_id": drug_id, "label": label.strip(), "filename": clean_filename, "sha256": digest})
    return cursor.lastrowid


def add_comparison(connection, old_id, new_id, actor):
    if old_id == new_id:
        raise HTTPException(422, "请选择两个不同版本")
    old = required_row(connection, "versions", old_id)
    new = required_row(connection, "versions", new_id)
    if old["drug_id"] != new["drug_id"]:
        raise HTTPException(422, "只能比对同一药品档案下的版本")
    if old["version_date"] > new["version_date"]:
        raise HTTPException(422, "旧版本日期不能晚于新版本日期")
    try:
        diff = compare_sections(json.loads(old["sections"]), json.loads(new["sections"]))
    except ValueError as error:
        raise HTTPException(422, str(error))
    cursor = connection.execute(
        """INSERT INTO comparisons(drug_id,old_version_id,new_version_id,unchanged_sections,warnings,created_at)
        VALUES(?,?,?,?,?,?)""", (old["drug_id"], old_id, new_id, diff["unchanged_sections"], encode(diff.get("warnings", [])), now())
    )
    comparison_id = cursor.lastrowid
    for change in diff["changes"]:
        connection.execute("INSERT INTO changes(comparison_id,payload) VALUES(?,?)", (comparison_id, encode(change)))
    audit(connection, actor, "创建版本比对", "comparison", comparison_id,
          {"old_version_id": old_id, "new_version_id": new_id, "changes": len(diff["changes"])})
    return comparison_id


def create_app(data_dir=None):
    application = FastAPI(
        title="药品说明书版本比对与变更审核系统", version="1.0.0",
        description="本地单操作员版本比对工具。输出用于人工复核，不判断医学正确性。",
    )
    db = Database(data_dir or os.environ.get("MEDLABEL_DATA_DIR", "data"))
    actor = os.environ.get("MEDLABEL_REVIEWER", "本地审核员").strip()[:200] or "本地审核员"
    allowed_hosts = {"localhost", "127.0.0.1", "::1"}
    # TestClient's synthetic hostname is only enabled for explicit app factories.
    if data_dir is not None:
        allowed_hosts.add("testserver")
    allowed_hosts.update(host.strip().lower() for host in os.environ.get("MEDLABEL_ALLOWED_HOSTS", "").split(",") if host.strip())
    application.state.db = db
    application.state.reviewer = actor

    @application.middleware("http")
    async def guard_requests(request: Request, call_next):
        # DNS rebinding must not make an attacker's origin a trusted local UI.
        host = request.headers.get("host", "")
        try:
            authority = urlsplit("//" + host)
            host_port = authority.port or (443 if request.url.scheme == "https" else 80)
            if authority.username or authority.password or authority.path or authority.query or authority.fragment:
                raise ValueError("Invalid authority")
            if (authority.hostname or "").lower() not in allowed_hosts:
                return JSONResponse({"detail": "不允许的访问主机，请使用 localhost 或 127.0.0.1"}, status_code=400)
        except ValueError:
            return JSONResponse({"detail": "无效的访问主机"}, status_code=400)
        if request.method in {"POST", "PUT", "PATCH", "DELETE"}:
            if request.headers.get("X-Requested-With") != "MedLabelReview":
                return JSONResponse({"detail": "请求缺少本地应用校验头"}, status_code=403)
            origin = request.headers.get("origin")
            if origin:
                try:
                    parsed_origin = urlsplit(origin)
                    origin_port = parsed_origin.port or (443 if parsed_origin.scheme == "https" else 80)
                    same_origin = (parsed_origin.scheme == request.url.scheme
                                   and parsed_origin.hostname == authority.hostname
                                   and origin_port == host_port
                                   and not parsed_origin.username and not parsed_origin.password
                                   and not parsed_origin.path and not parsed_origin.query and not parsed_origin.fragment)
                except ValueError:
                    same_origin = False
                if not same_origin:
                    return JSONResponse({"detail": "不允许跨来源修改本地数据"}, status_code=403)
            length = request.headers.get("content-length")
            if length:
                try:
                    if int(length) > MAX_FILE_SIZE + 256 * 1024:
                        return JSONResponse({"detail": "请求内容过大，文件不能超过 10 MB"}, status_code=413)
                except ValueError:
                    return JSONResponse({"detail": "无效的请求长度"}, status_code=400)
        response = await call_next(request)
        response.headers["X-Content-Type-Options"] = "nosniff"
        response.headers["X-Frame-Options"] = "DENY"
        response.headers["Referrer-Policy"] = "same-origin"
        response.headers["Cache-Control"] = "no-store"
        return response

    @application.exception_handler(RequestValidationError)
    async def validation_error(request, error):
        fields = [".".join(str(part) for part in item["loc"]) for item in error.errors()]
        return JSONResponse({"detail": "输入格式不正确，请检查：" + "、".join(fields)}, status_code=422)

    @application.get("/api/health")
    def health():
        return {"status": "ok", "version": "1.0.0", "mode": "local-single-operator", "reviewer": actor}

    @application.get("/api/dashboard")
    def dashboard():
        with db.connect() as connection:
            result = {table: connection.execute(f"SELECT COUNT(*) FROM {table}").fetchone()[0]
                      for table in ("drugs", "versions", "comparisons")}
            result["pending_changes"] = connection.execute("SELECT COUNT(*) FROM changes WHERE status='pending'").fetchone()[0]
            result["recent_activity"] = [dict(row) for row in connection.execute("SELECT * FROM audit ORDER BY id DESC LIMIT 8")]
            return result

    @application.get("/api/drugs")
    def list_drugs():
        with db.connect() as connection:
            return [dict(row) for row in connection.execute(
                "SELECT d.*, (SELECT COUNT(*) FROM versions v WHERE v.drug_id=d.id) AS version_count FROM drugs d ORDER BY d.id DESC")]

    @application.post("/api/drugs", status_code=201)
    def create_drug(payload: DrugInput):
        with db.connect(write=True) as connection:
            entity_id = add_drug(connection, payload.model_dump(), actor)
            result = dict(required_row(connection, "drugs", entity_id))
            result["version_count"] = 0
            return result

    @application.get("/api/drugs/{drug_id}")
    def get_drug(drug_id: int):

            result = dict(required_row(connection, "drugs", drug_id))
            result["version_count"] = connection.execute("SELECT COUNT(*) FROM versions WHERE drug_id=?", (drug_id,)).fetchone()[0]
            return result

    @application.get("/api/drugs/{drug_id}/versions")
    def list_versions(drug_id: int):
        with db.connect() as connection:
            required_row(connection, "drugs", drug_id)
            return [version_data(row) for row in connection.execute(
                "SELECT * FROM versions WHERE drug_id=? ORDER BY version_date DESC,id DESC", (drug_id,))]

    @application.post("/api/drugs/{drug_id}/versions", status_code=201)
    async def upload_version(drug_id: int, file: UploadFile = File(...), label: str = Form(...),
                             version_date: str = Form(...), source: str = Form("")):
        created_files = []
        try:
            data = await file.read(MAX_FILE_SIZE + 1)
            with db.connect(write=True) as connection:
                entity_id = add_version(connection, db, drug_id, label, version_date, source,
                                        file.filename or "", data, actor, created_files)
                return version_data(required_row(connection, "versions", entity_id), True)
        except Exception:
            for path in created_files:
                path.unlink(missing_ok=True)
            raise
        finally:
            await file.close()

    @application.post("/api/drugs/{drug_id}/versions/text", status_code=201)
    def text_version(drug_id: int, payload: TextVersionInput):
        created_files = []
        try:
            with db.connect(write=True) as connection:
                entity_id = add_version(connection, db, drug_id, payload.label, payload.version_date,
                                        payload.source, "pasted-text.txt", payload.text.encode("utf-8"), actor, created_files)
                return version_data(required_row(connection, "versions", entity_id), True)
        except Exception:
            for path in created_files:
                path.unlink(missing_ok=True)
            raise

    @application.get("/api/versions/{version_id}")
    def get_version(version_id: int):
        with db.connect() as connection:
            return version_data(required_row(connection, "versions", version_id), True)

    @application.get("/api/versions/{version_id}/file")
    def download_version(version_id: int):
        with db.connect() as connection:
            row = required_row(connection, "versions", version_id)
            path = (db.uploads / row["storage_name"]).resolve()
            if path.parent != db.uploads.resolve() or not path.is_file():
                raise HTTPException(404, "原始文件不存在")
            return FileResponse(path, filename=row["filename"], media_type="application/octet-stream")

    @application.put("/api/versions/{version_id}/sections")
    def update_sections(version_id: int, payload: SectionsInput):
        if sum(len(item.title) + len(item.text) for item in payload.sections) > MAX_TEXT:
            raise HTTPException(422, "章节总长度不能超过 300000 个字符")
        with db.connect(write=True) as connection:
            row = required_row(connection, "versions", version_id)
            linked = connection.execute("SELECT id FROM comparisons WHERE old_version_id=? OR new_version_id=? LIMIT 1",
                                        (version_id, version_id)).fetchone()
            if linked:
                raise HTTPException(409, "此版本已用于比对，不能修改章节。请导入一个新版本进行校正。")
            try:
                sections = normalize_sections([item.model_dump() for item in payload.sections])
            except ValueError as error:
                raise HTTPException(422, str(error))
            for item in sections:
                item["start_line"] = None
                item["end_line"] = None
            warnings = ["章节经人工校正；原始文件和提取原文保持不变，章节位置不再等同于原文行号。"]
            connection.execute("UPDATE versions SET sections=?,warnings=? WHERE id=?", (encode(sections), encode(warnings), version_id))
            audit(connection, actor, "校正章节", "version", version_id,
                  {"before": json.loads(row["sections"]), "after": sections})
            return version_data(required_row(connection, "versions", version_id), True)

    @application.get("/api/comparisons")
    def list_comparisons():
        with db.connect() as connection:
            ids = [row[0] for row in connection.execute("SELECT id FROM comparisons ORDER BY id DESC")]
            return [comparison_data(connection, entity_id, False) for entity_id in ids]

    @application.post("/api/comparisons", status_code=201)
    def create_comparison(payload: ComparisonInput):
        with db.connect(write=True) as connection:
            entity_id = add_comparison(connection, payload.old_version_id, payload.new_version_id, actor)
            return comparison_data(connection, entity_id)

    @application.get("/api/comparisons/{comparison_id}")
    def get_comparison(comparison_id: int):
        with db.connect() as connection:
            return comparison_data(connection, comparison_id)

    @application.patch("/api/changes/{change_id}")
    def review_change(change_id: int, payload: ReviewInput):
        if payload.status == "ignored" and not payload.note:
            raise HTTPException(422, "忽略差异时必须填写原因")
        with db.connect(write=True) as connection:
            current = required_row(connection, "changes", change_id)
            if current["revision"] != payload.revision:
                raise HTTPException(409, "此差异已被更新，请刷新后重新审核")
            timestamp = now()
            connection.execute(
                "UPDATE changes SET status=?,note=?,reviewer=?,reviewed_at=?,revision=revision+1 WHERE id=? AND revision=?",
                (payload.status, payload.note, actor, timestamp, change_id, payload.revision),
            )
            audit(connection, actor, "审核变更", "change", change_id,
                  {"comparison_id": current["comparison_id"], "before": {"status": current["status"], "note": current["note"]},
                   "after": {"status": payload.status, "note": payload.note}, "revision": payload.revision + 1})
            return change_data(required_row(connection, "changes", change_id))

    @application.get("/api/audit")
    def list_audit():
        with db.connect() as connection:
            return [dict(row) for row in connection.execute("SELECT * FROM audit ORDER BY id DESC LIMIT 500")]

    @application.get("/api/comparisons/{comparison_id}/export")
    def export_comparison(comparison_id: int, format: Literal["html", "csv", "json"] = "html"):

            comparison = comparison_data(connection, comparison_id)
        body, media_type, filename = export_report(comparison, format)
        return Response(body, media_type=media_type, headers={"Content-Disposition": f'attachment; filename="{filename}"'})

                old_id = add_version(connection, db, drug_id, "模拟初始版", "2026-01-01", "项目内置虚构示例，不用于临床",
                                     "demo-v1.txt", OLD_TEXT.encode("utf-8"), actor, created_files)
                new_id = add_version(connection, db, drug_id, "模拟修订版", "2026-06-01", "项目内置虚构示例，不用于临床",
                                     "demo-v2.txt", NEW_TEXT.encode("utf-8"), actor, created_files)
                comparison_id = add_comparison(connection, old_id, new_id, actor)
                result = {"drug_id": drug_id, "comparison_id": comparison_id}
                connection.execute("INSERT INTO settings(key,value) VALUES('demo',?)", (encode(result),))
                return result
        except Exception:
            for path in created_files:
                path.unlink(missing_ok=True)
            raise

    def index():

        return FileResponse(path)

    application.mount("/static", StaticFiles(directory=STATIC_DIR, check_dir=False), name="static")
    return application

"""SQLite persistence; every mutation and its audit record share a transaction."""
from contextlib import contextmanager
from datetime import datetime, timezone
import json
from pathlib import Path
import sqlite3


def now():
    return datetime.now(timezone.utc).isoformat(timespec="microseconds")


def encode(value):
    return json.dumps(value, ensure_ascii=False)


SCHEMA = """
CREATE TABLE IF NOT EXISTS drugs (
 id INTEGER PRIMARY KEY, name TEXT NOT NULL, manufacturer TEXT NOT NULL DEFAULT '',
 specification TEXT NOT NULL DEFAULT '', approval_number TEXT NOT NULL DEFAULT '',
 notes TEXT NOT NULL DEFAULT '', created_at TEXT NOT NULL
);
CREATE TABLE IF NOT EXISTS versions (
 id INTEGER PRIMARY KEY, drug_id INTEGER NOT NULL REFERENCES drugs(id),
 label TEXT NOT NULL, version_date TEXT NOT NULL, source TEXT NOT NULL DEFAULT '',
 filename TEXT NOT NULL, sha256 TEXT NOT NULL, storage_name TEXT NOT NULL,
 text TEXT NOT NULL, sections TEXT NOT NULL, warnings TEXT NOT NULL DEFAULT '[]',
 created_at TEXT NOT NULL
);
CREATE TABLE IF NOT EXISTS comparisons (
 id INTEGER PRIMARY KEY, drug_id INTEGER NOT NULL REFERENCES drugs(id),
 old_version_id INTEGER NOT NULL REFERENCES versions(id),
 new_version_id INTEGER NOT NULL REFERENCES versions(id),
 unchanged_sections INTEGER NOT NULL, warnings TEXT NOT NULL DEFAULT '[]',
 created_at TEXT NOT NULL,
 CHECK(old_version_id != new_version_id)
);
CREATE TABLE IF NOT EXISTS changes (
 id INTEGER PRIMARY KEY, comparison_id INTEGER NOT NULL REFERENCES comparisons(id),
 payload TEXT NOT NULL, status TEXT NOT NULL DEFAULT 'pending'
 CHECK(status IN ('pending','confirmed','ignored')),
 note TEXT NOT NULL DEFAULT '', reviewer TEXT NOT NULL DEFAULT '',
 reviewed_at TEXT, revision INTEGER NOT NULL DEFAULT 0
);
CREATE TABLE IF NOT EXISTS audit (
 id INTEGER PRIMARY KEY, action TEXT NOT NULL, entity_type TEXT NOT NULL,
 entity_id INTEGER NOT NULL, detail TEXT NOT NULL, actor TEXT NOT NULL,
 created_at TEXT NOT NULL
);
CREATE TABLE IF NOT EXISTS settings (key TEXT PRIMARY KEY, value TEXT NOT NULL);
CREATE INDEX IF NOT EXISTS versions_drug ON versions(drug_id);
CREATE INDEX IF NOT EXISTS changes_comparison ON changes(comparison_id);
CREATE INDEX IF NOT EXISTS comparisons_versions ON comparisons(old_version_id,new_version_id);
"""


class Database:
    def __init__(self, data_dir):
        self.directory = Path(data_dir).resolve()
        self.directory.mkdir(parents=True, exist_ok=True)
        self.uploads = self.directory / "uploads"
        self.uploads.mkdir(exist_ok=True)
        self.path = self.directory / "medlabel.sqlite3"
        with self.connect() as connection:
            connection.executescript(SCHEMA)

    @contextmanager
    def connect(self, write=False):
        connection = sqlite3.connect(self.path, timeout=15)
        connection.row_factory = sqlite3.Row
        connection.execute("PRAGMA foreign_keys = ON")
        connection.execute("PRAGMA journal_mode = WAL")
        try:
            if write:
                connection.execute("BEGIN IMMEDIATE")
            yield connection
            connection.commit()
        except Exception:
            connection.rollback()
            raise
        finally:
            connection.close()


def audit(connection, actor, action, entity_type, entity_id, detail):
    connection.execute(
        "INSERT INTO audit(action,entity_type,entity_id,detail,actor,created_at) VALUES(?,?,?,?,?,?)",
        (action, entity_type, entity_id, encode(detail) if not isinstance(detail, str) else detail, actor, now()),
    )

"""Extract and structure drug leaflets without inferring medical meaning.

Only user supplied text is processed.  The original file/text is retained by the
API; these sections are a reviewable representation, never a replacement source.
"""

from __future__ import annotations

import io
import re
import unicodedata
import zipfile
from collections import Counter
from pathlib import PurePath


MAX_FILE_BYTES = 10 * 1024 * 1024
MAX_TEXT_CHARS = 1_000_000
MAX_PDF_PAGES = 100
MAX_SECTIONS = 500
MAX_DOCX_EXPANDED = 40 * 1024 * 1024

_BRACKET_HEADING = re.compile(r"^\s*(?:【([^】\n]{1,60})】|\[([^\]\n]{1,60})\]|［([^］\n]{1,60})］)\s*[:：]?\s*(.*)$")
_NUMBER_PREFIX = re.compile(r"^\s*(?:(?:[一二三四五六七八九十百]+|\d+)[、.．]\s*|[（(](?:[一二三四五六七八九十百]+|\d+)[）)]\s*)")


def _clean_text(text: str) -> str:
    text = text.replace("\r\n", "\n").replace("\r", "\n").replace("\ufeff", "")
    if "\x00" in text:
        raise ValueError("文档包含二进制或空字符，请转换为文字版文件后重试")
    if len(text) > MAX_TEXT_CHARS:
        raise ValueError("文档文字超过 100 万字符，请拆分为单份药品说明书")
    if not text.strip():
        raise ValueError("未提取到可用文字；扫描件需要先进行 OCR 并人工核对")
    return text


def _pdf_contains_image(page, reader) -> bool:
    """Inspect resource graphs and parsed operators, never text-density guesses.

    PDF images can coexist with a tiny text layer (for example only a page
    number), be nested in Form XObjects, or be embedded as inline images. V1
    deliberately rejects all such pages, including otherwise harmless logos.
    """
    from pypdf.generic import ContentStream

    seen_resources: set[int] = set()
    seen_streams: set[int] = set()

    def resolve(value):
        return value.get_object() if hasattr(value, "get_object") else value

    def inspect(stream, resources, depth=0):
        if depth > 32 or len(seen_resources) + len(seen_streams) > 2000:
            raise ValueError("PDF 图形结构过于复杂，无法确认文字完整性；请转换为完整文字并核对")
        stream = resolve(stream)
        resources = resolve(resources) if resources else {}
        if stream is not None and id(stream) not in seen_streams:
            seen_streams.add(id(stream))
            parsed = stream if isinstance(stream, ContentStream) else ContentStream(stream, reader)
            if any(operator == b"INLINE IMAGE" for _, operator in parsed.operations):
                return True
        if id(resources) in seen_resources:
            return False
        seen_resources.add(id(resources))
        for category in ("/XObject", "/Pattern"):
            objects = resolve(resources.get(category, {}))
            for reference in objects.values():
                obj = resolve(reference)
                if obj.get("/Subtype") == "/Image":
                    return True
                if obj.get("/Subtype") == "/Form" or obj.get("/PatternType") == 1:
                    if inspect(obj, obj.get("/Resources", resources), depth + 1):
                        return True
        return False

    resources = page.get("/Resources", {})
    if inspect(page.get_contents(), resources):
        return True
    # Annotation appearance streams may also contain images not referenced by
    # the main page content stream (for example a scanned stamp or attachment).
    for reference in resolve(page.get("/Annots", [])):
        annotation = resolve(reference)
        appearances = resolve(annotation.get("/AP", {}))
        for appearance in appearances.values():
            appearance = resolve(appearance)
            candidates = [appearance] if hasattr(appearance, "get_data") else [resolve(value) for value in appearance.values()]
            for candidate in candidates:
                if inspect(candidate, candidate.get("/Resources", resources)):
                    return True
    return False


def _pdf_text(data: bytes) -> str:
    from pypdf import PdfReader

    reader = PdfReader(io.BytesIO(data), strict=False)
    if reader.is_encrypted:
        raise ValueError("PDF 已加密，请提供解密后的文字版 PDF")
    if len(reader.pages) > MAX_PDF_PAGES:
        raise ValueError("PDF 超过 100 页，请拆分为单份药品说明书")
    texts: list[str] = []
    length = 0
    for index, page in enumerate(reader.pages, start=1):
        if _pdf_contains_image(page, reader):
            raise ValueError(f"此 PDF 含图片内容，可能遗漏文字；请先转换为完整文字并核对（第 {index} 页）")
        text = page.extract_text() or ""
        if text.strip():
            texts.append(text)
            length += len(text) + 2
        if length > MAX_TEXT_CHARS:
            raise ValueError("文档文字超过 100 万字符，请拆分为单份药品说明书")
    return "\n\n".join(texts)


def _docx_text(data: bytes) -> str:
    from docx import Document
    from docx.oxml.ns import qn

    with zipfile.ZipFile(io.BytesIO(data)) as archive:
        members = archive.infolist()
        if len(members) > 2500 or sum(member.file_size for member in members) > MAX_DOCX_EXPANDED:
            raise ValueError("DOCX 解压后体积或文件数量过大，请简化文件后重试")
        if any(member.flag_bits & 1 for member in members):
            raise ValueError("不支持加密的 DOCX 文件")
        if any(member.file_size > 1024 * 1024 and member.file_size / max(member.compress_size, 1) > 200 for member in members):
            raise ValueError("DOCX 压缩比例异常，请重新保存为普通文档")
        if "word/document.xml" not in archive.namelist():
            raise ValueError("文件不是有效的 DOCX 文档")

    document = Document(io.BytesIO(data))
    headers, footers = [], []
    seen_parts: set[str] = set()
    for relation in document.part.rels.values():
        if relation.is_external or not relation.reltype.endswith(("/header", "/footer")):
            continue
        part = relation.target_part
        if str(part.partname) in seen_parts:
            continue
        seen_parts.add(str(part.partname))
        (headers if relation.reltype.endswith("/header") else footers).append(part.element)
    roots = headers + [document.element.body] + footers
    for root in roots:
        if root.xpath(".//w:footnoteReference | .//w:endnoteReference"):
            raise ValueError("DOCX 含脚注或尾注，请先转换为包含全部文字的 TXT，避免遗漏")
        if root.xpath(".//w:ins | .//w:del"):
            raise ValueError("DOCX 含未处理的修订记录，请确认修订后重新上传")
    # Iterate paragraphs in XML document order. This includes paragraphs inside
    # table cells and text boxes, which Document.paragraphs alone would omit.
    paragraphs: list[str] = []
    for root in roots:
        if root in headers:
            paragraphs.append("【页眉内容】")
        elif root in footers:
            paragraphs.append("【页脚内容】")
        for paragraph in root.iter(qn("w:p")):
            fragments: list[str] = []
            for node in paragraph.iter():
                # Text-box paragraphs are visited in their own iteration; skip
                # their text here to avoid duplication in the outer paragraph.
                parent = node.getparent()
                in_nested_paragraph = False
                while parent is not None and parent is not paragraph:
                    if parent.tag == qn("w:p"):
                        in_nested_paragraph = True
                        break
                    parent = parent.getparent()
                if in_nested_paragraph:
                    continue
                if node.tag == qn("w:t") and node.text:
                    fragments.append(node.text)
                elif node.tag == qn("w:tab"):
                    fragments.append("\t")
                elif node.tag in {qn("w:br"), qn("w:cr")}:
                    fragments.append("\n")
            paragraphs.append("".join(fragments))
    return "\n".join(paragraphs)


def extract_text(filename: str, data: bytes) -> str:
    """Extract a supported document or raise a user-facing ``ValueError``.

    Filenames are used only to select format, never as filesystem destinations.
    Size checks are duplicated here so callers other than HTTP cannot bypass them.
    """
    if not isinstance(data, bytes) or not data:
        raise ValueError("文件为空，请选择包含文字的文件")
    if len(data) > MAX_FILE_BYTES:
        raise ValueError("文件超过 10 MB，请压缩或拆分后重试")
    extension = PurePath(filename.replace("\\", "/")).suffix.lower()
    try:
        if extension == ".txt":
            text = None
            for encoding in ("utf-8-sig", "gb18030"):
                try:
                    text = data.decode(encoding)
                    break
                except UnicodeDecodeError:
                    pass
            if text is None:
                raise ValueError("TXT 编码无法识别，请另存为 UTF-8")
        elif extension == ".pdf":
            if not data.lstrip().startswith(b"%PDF-"):
                raise ValueError("文件内容不是有效的 PDF")
            text = _pdf_text(data)
        elif extension == ".docx":
            text = _docx_text(data)
        else:
            raise ValueError("仅支持 TXT、文字版 PDF 和 DOCX 文件；旧版 DOC 请另存为 DOCX")
        return _clean_text(text)
    except ValueError:
        raise
    except Exception as exc:
        raise ValueError("文档无法解析，可能已损坏或格式不匹配；请重新导出为文字版文件") from exc


def canonical_title(title: str) -> str:
    """Normalize only heading formatting and explicitly documented aliases."""
    normalized = unicodedata.normalize("NFKC", title).strip().strip("【】[]［］:： ")
    normalized = _NUMBER_PREFIX.sub("", normalized)
    compact = re.sub(r"\s+", "", normalized)
    return _ALIASES.get(compact, compact or "未命名章节")


def _heading(line: str) -> tuple[str, str] | None:
    match = _BRACKET_HEADING.match(line)
    if match:
        title = next(part for part in match.groups()[:3] if part is not None).strip()
        # [1] at the start of a reference paragraph is not a section title.
        if re.fullmatch(r"[\d\W_]+", title):
            return None
        return title, match.group(4)
    stripped = _NUMBER_PREFIX.sub("", line.strip())
    # Unbracketed headings must exactly match a known heading, or be followed
    # by a colon. Arbitrary short body sentences are never promoted to headings.
    title, separator, remainder = stripped.partition("：")
    if not separator:
        title, separator, remainder = stripped.partition(":")
    compact = re.sub(r"\s+", "", title)
    if compact in _ALIASES:
        return title.strip(), remainder.strip() if separator else ""
    return None


def _assign_keys(sections: list[dict]) -> list[dict]:
    counts: Counter[str] = Counter()
    for section in sections:
        # Reserve ASCII '#' for the occurrence suffix; literal title characters
        # must not collide with generated keys such as 成份#2.
        canonical = canonical_title(section["title"]).replace("#", "＃")
        counts[canonical] += 1
        section["key"] = canonical if counts[canonical] == 1 else f"{canonical}#{counts[canonical]}"
    return sections


def parse_sections(text: str) -> list[dict]:
    """Split sections, preserving preamble, duplicates, and source line ranges.

    Line numbers refer to extracted text, not PDF page layout. Unknown bracketed
    headings remain separate sections. Unstructured documents become one 全文
    section so content cannot silently disappear.
    """
    text = _clean_text(text)
    lines = text.split("\n")
    sections: list[dict] = []
    current: dict | None = None
    buffer: list[str] = []

    def flush(end_line: int) -> None:
        if current is None:
            return
        current["text"] = "\n".join(buffer).strip()
        current["end_line"] = max(current["start_line"], end_line)
        sections.append(current)

    for number, line in enumerate(lines, start=1):
        heading = _heading(line)
        if heading is not None:
            flush(number - 1)
            title, remainder = heading
            current = {"title": title, "start_line": number}
            buffer = [remainder] if remainder else []
        else:
            if current is None:
                if not line.strip():
                    continue
                current = {"title": "前言", "start_line": number}
            buffer.append(line)
    flush(len(lines))
    if len(sections) == 1 and sections[0]["title"] == "前言":
        sections[0]["title"] = "全文"
    if len(sections) > MAX_SECTIONS:
        raise ValueError("章节数量超过 500，请检查文件格式或拆分文档")
    return _assign_keys(sections)


def normalize_sections(sections: list[dict]) -> list[dict]:
    """Validate a manual section correction and regenerate stable keys.

    Source line numbers are intentionally cleared: manual corrections cannot
    honestly claim exact correspondence with extraction line positions.
    """
    if not isinstance(sections, list) or not sections:
        raise ValueError("至少保留一个章节")
    if len(sections) > MAX_SECTIONS:
        raise ValueError("章节数量不能超过 500")
    result: list[dict] = []
    total = 0
    for section in sections:
        if not isinstance(section, dict):
            raise ValueError("章节格式错误")
        title = section.get("title")
        text = section.get("text")
        if not isinstance(title, str) or not title.strip() or len(title.strip()) > 120:
            raise ValueError("章节标题应为 1 至 120 个字符")
        if not isinstance(text, str) or "\x00" in text:
            raise ValueError("章节内容应为有效文本")
        text = text.replace("\r\n", "\n").replace("\r", "\n").strip()
        total += len(text)
        result.append({"title": title.strip(), "text": text, "start_line": None, "end_line": None})
    if total > MAX_TEXT_CHARS:
        raise ValueError("章节文字总量不能超过 100 万字符")
    if not any(section["text"] for section in result):
        raise ValueError("章节内容不能全部为空")
    return _assign_keys(result)

"""Deterministic section alignment and review prioritization.

Priority and flags identify textual review tasks, not clinical risk or a medical
recommendation. Inline segments are raw text and MUST be escaped by renderers.
"""

from __future__ import annotations

import re
from collections import Counter
from difflib import SequenceMatcher

from .documents import canonical_title


_IMPORTANT_SECTIONS = {
    "用法用量", "禁忌", "不良反应", "注意事项", "警告", "黑框警告",
    "孕妇及哺乳期妇女用药", "儿童用药", "老年用药", "特殊人群用药",
    "药物相互作用", "药物过量", "适应症", "功能主治",
}
_NUMBER = re.compile(r"(?<![A-Za-z0-9.])[-+]?\d+(?:[.,]\d+)*(?:\s*[%％])?")
_UNIT = re.compile(r"(?<![A-Za-z])(?:mcg|μg|µg|mg|kg|g|mL|ml|L|IU|mmol|mol|μL|µL|℃|°C|毫克|微克|克|毫升|升|单位|片|粒|支|袋|滴)(?![A-Za-z])", re.IGNORECASE)


def _comparison_text(text: str) -> str:
    """Ignore layout whitespace without merging distinct numeric tokens.

    In particular '1 0 mg' never becomes '10 mg'; decimal points, units,
    punctuation and negation remain significant.
    """
    text = re.sub(r"\s+", " ", text).strip()
    text = re.sub(r"(?<=[\u3400-\u9fff]) +(?=[\u3400-\u9fff])", "", text)
    return text


def _indexed(sections: list[dict]) -> tuple[dict[str, dict], list[str]]:
    counts: Counter[str] = Counter()
    result: dict[str, dict] = {}
    order: list[str] = []
    # Recompute keys from headings rather than trusting mutable caller keys.
    for section in sections:
        title = canonical_title(section["title"]).replace("#", "＃")
        counts[title] += 1
        key = title if counts[title] == 1 else f"{title}#{counts[title]}"
        result[key] = section
        order.append(key)
    return result, order


def _location(section: dict | None) -> str:
    if section is None:
        return "无对应章节"
    start, end = section.get("start_line"), section.get("end_line")
    if not isinstance(start, int) or not isinstance(end, int):
        return "手工校正章节"
    return f"提取文本第 {start} 行" if start == end else f"提取文本第 {start}–{end} 行"


def _inline_segments(old: str, new: str) -> tuple[list[dict], bool]:
    if not old:
        return ([{"op": "insert", "text": new}] if new else []), False
    if not new:
        return [{"op": "delete", "text": old}], False
    # Character-level matching is useful for Chinese units and negations, but
    # huge or pathological strings need a bounded whole-section fallback.
    if len(old) + len(new) > 30_000:
        return [{"op": "delete", "text": old}, {"op": "insert", "text": new}], True
    matcher = SequenceMatcher(None, old, new, autojunk=True)
    segments: list[dict] = []
    for op, a_start, a_end, b_start, b_end in matcher.get_opcodes():
        if op == "equal":
            segments.append({"op": "equal", "text": old[a_start:a_end]})
        else:
            if op in {"delete", "replace"}:
                segments.append({"op": "delete", "text": old[a_start:a_end]})
            if op in {"insert", "replace"}:
                segments.append({"op": "insert", "text": new[b_start:b_end]})
    return segments, False


def compare_sections(old_sections: list[dict], new_sections: list[dict]) -> dict:
    """Compare each canonical heading, pairing repeated headings by occurrence.

    Reordering alone is reported as a warning, not falsely presented as new
    clinical content. Differences are returned in new-document order, followed
    by removed sections in old-document order.
    """
    old_map, old_order = _indexed(old_sections)
    new_map, new_order = _indexed(new_sections)
    warnings: list[str] = []
    if any("#" in key for key in old_order + new_order):
        warnings.append("存在重复标题，已按同名章节出现顺序配对；请人工核对章节对应关系。")
    if "全文" in old_map or "全文" in new_map:
        warnings.append("至少一个版本未识别出标准章节；可在创建比对前手工整理章节，以提高定位精度。")
    shared_old = [key for key in old_order if key in new_map]
    shared_new = [key for key in new_order if key in old_map]
    if shared_old != shared_new:
        warnings.append("章节顺序发生变化；已按标题对齐，顺序调整本身不计为内容变更。")
    changes: list[dict] = []
    unchanged = 0
    all_keys = new_order + [key for key in old_order if key not in new_map]
    for key in all_keys:
        old, new = old_map.get(key), new_map.get(key)
        old_text = old["text"] if old else ""
        new_text = new["text"] if new else ""
        if old is not None and new is not None and _comparison_text(old_text) == _comparison_text(new_text):
            unchanged += 1
            continue
        kind = "added" if old is None else "removed" if new is None else "modified"
        section = new if new is not None else old
        title = section["title"]
        canonical = canonical_title(title)
        flags: list[str] = []
        if canonical in _IMPORTANT_SECTIONS:
            flags.append("重点章节变更，需人工复核")
        if _NUMBER.findall(old_text) != _NUMBER.findall(new_text):
            flags.append("数字或比例变化")
        if _UNIT.findall(old_text) != _UNIT.findall(new_text):
            flags.append("单位或剂型计量词变化")
        segments, coarse = _inline_segments(old_text, new_text)
        if coarse:
            flags.append("长章节采用整段展示，请对照原文复核")
        changes.append({
            "section_key": key,
            "section_title": title,
            "kind": kind,
            "priority": "high" if canonical in _IMPORTANT_SECTIONS or any("变化" in flag for flag in flags) else "normal",
            "old_text": old_text,
            "new_text": new_text,
            "old_location": _location(old),
            "new_location": _location(new),
            "segments": segments,
            "flags": flags,
        })
    return {"changes": changes, "unchanged_sections": unchanged, "warnings": warnings}

function fmtDate(value, includeTime = false) {
  if (!value) return "未记录";
  const date = new Date(value);
  if (Number.isNaN(date.getTime())) return String(value);
  return new Intl.DateTimeFormat("zh-CN", {
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
    ...(includeTime ? { hour: "2-digit", minute: "2-digit" } : {}),
  }).format(date);
}

async function api(path, options = {}) {
  const headers = new Headers(options.headers || {});
  if (options.method && options.method !== "GET")
    headers.set("X-Requested-With", "MedLabelReview");
  if (options.body && !(options.body instanceof FormData)) {
    headers.set("Content-Type", "application/json");
    options.body = JSON.stringify(options.body);
  }
  let response;
  try {
    response = await fetch(path, {
      ...options,
      headers,
      credentials: "same-origin",
    });
  } catch (_) {
    throw new Error("无法连接本地服务，请确认应用已启动后重试。");
  }
  let result;
  try {
    result = await response.json();
  } catch (_) {
    throw new Error("服务返回了无法读取的内容，请稍后重试。");
  }
  if (!response.ok) {
    let detail =
      result.detail || result.message || `请求失败（${response.status}）`;
    if (Array.isArray(detail))
      detail = detail
        .map(
          (item) =>
            `${(item.loc || []).filter((p) => p !== "body").join(" / ")}：${item.msg}`,
        )
        .join("；");
    const error = new Error(
      typeof detail === "string" ? detail : JSON.stringify(detail),
    );
    error.status = response.status;
    throw error;
  }
  return result;
}

async function navigate() {
  const route = location.hash.slice(1) || "dashboard";
  const generation = ++state.generation;
  const previous = state.route;
  state.route = route;
  if (previous !== route) {
    state.reviewStatus = "all";
    state.priority = "all";
  }
  const parts = route.split("/");
  const section = ["dashboard", "drugs", "comparisons", "audit"].includes(
    parts[0],
  )
    ? parts[0]
    : "dashboard";
  $$("[data-nav]").forEach((element) => {
    const selected = element.dataset.nav === section;
    element.classList.toggle("active", selected);
    if (selected) element.setAttribute("aria-current", "page");
    else element.removeAttribute("aria-current");
  });
  $("#breadcrumb-current").textContent = {
    dashboard: "工作台",
    drugs: parts[1] ? "药品档案" : "药品与版本",
    comparisons: parts[1] ? "变更审核" : "比对与审核",
    audit: "操作记录",
  }[section];
  main.innerHTML =
    '<div class="loading-state"><span class="spinner"></span>正在加载…</div>';
  try {
    let html;
    if (section === "drugs" && parts[1])
      html = await drugPage(Number(parts[1]));
    else if (section === "drugs") html = await drugsPage();
    else if (section === "comparisons" && parts[1])
      html = await comparisonPage(Number(parts[1]));
    else if (section === "comparisons") html = await comparisonsPage();
    else if (section === "audit") html = await auditPage();
    else html = await dashboardPage();
    if (generation !== state.generation) return;
    main.innerHTML = html;
    document.title = `${$("#breadcrumb-current").textContent} · MedLabel Review`;
    if (previous !== route) window.scrollTo(0, 0);
  } catch (error) {
    if (generation === state.generation) renderError(error);
  }
}

function detailText(detail) {
  let value = detail;
  if (typeof value === "string") {
    try {
      value = JSON.parse(value);
    } catch (_) {
      return value;
    }
  }
  if (!value || typeof value !== "object") return String(value ?? "");
  if (Array.isArray(value.before) && Array.isArray(value.after)) {
    return `章节校正：${value.before.length} 个 → ${value.after.length} 个章节`;
  }
  if (value.before?.status && value.after?.status) {
    const labels = {
      pending: "待复核",
      confirmed: "已确认",
      ignored: "已忽略",
    };
    return `${labels[value.before.status] || value.before.status} → ${labels[value.after.status] || value.after.status}${value.after.note ? `；复核意见：${value.after.note}` : ""}`;
  }
  const labels = {
    name: "药品",
    label: "版本",
    filename: "文件",
    drug_id: "药品档案",
    old_version_id: "旧版编号",
    new_version_id: "新版编号",
    changes: "变更数",
    comparison_id: "比对编号",
    note: "复核意见",
    status: "状态",
  };
  return Object.entries(value)
    .filter(([key]) => key !== "sha256")
    .map(
      ([key, item]) =>
        `${labels[key] || key}：${typeof item === "object" ? JSON.stringify(item) : item}`,
    )
    .join(" · ");
}

function markedText(text) {
  // Match untrusted plain text first; escaping before matching could split an HTML entity.
  const value = String(text ?? "");
  const pattern =
    /\d+(?:\.\d+)?(?:\s*(?:mg|μg|ug|kg|g|ml|mL|L|%|毫克|微克|克|毫升|片|次|日|天))?/g;
  let result = "";
  let cursor = 0;
  for (const match of value.matchAll(pattern)) {
    result += esc(value.slice(cursor, match.index));
    result += `<span class="numeric">${esc(match[0])}</span>`;
    cursor = match.index + match[0].length;
  }
  return result + esc(value.slice(cursor));
}
function diffText(change, side) {
  const text = side === "old" ? change.old_text : change.new_text;
  if (!text) return '<span class="absent-text">该版本无此章节内容</span>';
  const segments = change.segments;
  if (!Array.isArray(segments) || !segments.length) return esc(text);
  const relevant = segments.filter(
    (segment) =>
      segment.op === "equal" ||
      segment.op === (side === "old" ? "delete" : "insert"),
  );
  // Only apply inline segments if they reconstruct this exact source; never hide source text.
  if (relevant.map((segment) => segment.text || "").join("") !== text)
    return esc(text);
  return relevant
    .map((segment) =>
      segment.op === "equal"
        ? esc(segment.text)
        : `<${side === "old" ? "del" : "ins"}>${markedText(segment.text)}</${side === "old" ? "del" : "ins"}>`,
    )
    .join("");
}
function changesMarkup() {
  const changes = state.comparison.changes || [];
  const filtered = changes.filter(
    (change) =>
      (state.reviewStatus === "all" || change.status === state.reviewStatus) &&
      (state.priority === "all" || change.priority === state.priority),
  );
  if (!filtered.length)
    return `<section class="panel">${empty(changes.length ? "当前筛选下没有变更" : "两个版本未发现章节内容差异", changes.length ? "切换状态或优先级，查看其他变更。" : "已保留本次比对记录，可导出报告。", "", true)}</section>`;
  return filtered
    .map((change) => {
      const index = changes.findIndex((item) => item.id === change.id) + 1;
      const note = state.drafts.has(change.id)
        ? state.drafts.get(change.id)
        : change.note || "";
      return `<article class="change-card" id="change-${id(change.id)}" data-priority="${change.priority === "high" ? "high" : "normal"}"><header class="change-card-head"><div class="change-title"><span class="change-index">${String(index).padStart(2, "0")}</span><h3>${esc(change.section_title)}</h3>${kindTag(change.kind)}${change.priority === "high" ? '<span class="tag amber">重点复核</span>' : ""}</div>${statusTag(change.status)}</header>${change.flags?.length ? `<div class="change-flags">${change.flags.map((flag) => `<span>${esc(flag)}</span>`).join("")}</div>` : ""}<div class="diff-grid"><div class="diff-column"><div class="diff-label"><span>旧版内容</span><span>${esc(change.old_location || "—")}</span></div><div class="diff-text">${diffText(change, "old")}</div></div><div class="diff-column"><div class="diff-label"><span>新版内容</span><span>${esc(change.new_location || "—")}</span></div><div class="diff-text">${diffText(change, "new")}</div></div></div><div class="review-area">${change.reviewed_at ? `<div class="review-history"><strong>${esc(change.reviewer || "本地审核员")}</strong> · ${esc(fmtDate(change.reviewed_at, true))} · 已保存上次审核记录</div>` : ""}<div class="review-controls"><div class="review-note"><label for="note-${id(change.id)}">复核意见 <span class="muted">（忽略变更时必填）</span></label><textarea id="note-${id(change.id)}" data-note-id="${id(change.id)}" maxlength="3000" rows="1" placeholder="记录变更依据或处理意见…">${esc(note)}</textarea></div><div class="actions">${change.status !== "pending" ? button("重置待复核", "review", `data-id="${id(change.id)}" data-status="pending"`) : ""}${button("忽略", "review", `data-id="${id(change.id)}" data-status="ignored"`)}${button(change.status === "confirmed" ? "保存确认意见" : "确认变更", "review", `data-id="${id(change.id)}" data-status="confirmed"`, true, "check")}</div></div></div></article>`;
    })
    .join("");
}

  if (element.dataset.noteId) {
    state.drafts.set(Number(element.dataset.noteId), element.value);
    return;
  }
  if (element.id === "drug-search") {
    const query = element.value.toLocaleLowerCase().trim();
    const filtered = state.drugs.filter((item) =>
      [item.name, item.manufacturer, item.approval_number].some((value) =>
        String(value || "")
          .toLocaleLowerCase()
          .includes(query),
      ),
    );
    $("#drug-list").innerHTML = drugTable(filtered);
    $("#drug-count").textContent = `${filtered.length} 个药品档案`;
  }
  if (element.id === "comparison-search") filterComparisonTable();
  if (element.id === "audit-search") {
    const query = element.value.toLocaleLowerCase().trim();
    const filtered = state.audit.filter((item) =>
      [
        auditLabel(item.action),
        item.entity_type,
        item.entity_id,
        item.actor,
        detailText(item.detail),
      ].some((value) =>
        String(value || "")
          .toLocaleLowerCase()
          .includes(query),
      ),
    );
    $("#audit-list").innerHTML = auditTable(filtered);
    $("#audit-count").textContent = `${filtered.length} 条记录`;
  }
});
function filterComparisonTable() {
  const query = $("#comparison-search").value.toLocaleLowerCase().trim();
  const status = $("#comparison-status").value;
  const filtered = state.comparisons.filter(
    (item) =>
      [item.drug_name, item.old_label, item.new_label].some((value) =>
        String(value || "")
          .toLocaleLowerCase()
          .includes(query),
      ) &&
      (status === "all" ||

          ? Number(item.pending_changes) > 0
          : !Number(item.pending_changes))),
  );
  $("#comparison-list").innerHTML = comparisonTable(filtered);
  $("#comparison-count").textContent = `${filtered.length} 个比对任务`;
}

document.addEventListener("click", async (event) => {
  const trigger = event.target.closest("[data-action]");
  if (!trigger || trigger.disabled) return;
  const action = trigger.dataset.action;
  try {
    if (action === "new-drug") return newDrugModal();
    if (action === "close-modal") return closeModal();
    if (action === "refresh") return navigate();
    if (action === "import-version") return importModal(trigger.dataset.id);
    if (action === "filter-review") {
      state.reviewStatus = trigger.dataset.value;
      $$('[data-action="filter-review"]').forEach((button) => {
        const active = button.dataset.value === state.reviewStatus;
        button.classList.toggle("active", active);
        button.setAttribute("aria-pressed", String(active));
      });
      $("#changes-list").innerHTML = changesMarkup();
      return;
    }
    if (action === "import-tab") {
      const mode = trigger.dataset.value;
      $("#import-mode").value = mode;
      $$('[data-action="import-tab"]').forEach((button) => {
        const active = button.dataset.value === mode;
        button.classList.toggle("active", active);
        button.setAttribute("aria-pressed", String(active));
      });
      $("#import-file-pane").hidden = mode !== "file";
      $("#import-text-pane").hidden = mode !== "text";
      $("#version-file").disabled = mode !== "file";
      $("#version-file").required = mode === "file";
      $("#version-text").disabled = mode !== "text";
      $("#version-text").required = mode === "text";
      return;
    }
    if (action === "version-tab") {
      const tab = trigger.dataset.value;
      $$('[data-action="version-tab"]').forEach((button) => {
        const active = button.dataset.value === tab;
        button.classList.toggle("active", active);
        button.setAttribute("aria-pressed", String(active));
      });
      ["sections", "raw", "source"].forEach((name) => {
        $(`#version-${name}-pane`).hidden = name !== tab;
      });

async function saveReview(trigger) {
  const change = state.comparison.changes.find(
    (item) => item.id === Number(trigger.dataset.id),
  );
  if (!change) throw new Error("变更记录已更新，请刷新页面。");
  const note = $(`#note-${id(change.id)}`).value.trim();
  const status = trigger.dataset.status;
  if (status === "ignored" && !note) {
    $(`#note-${id(change.id)}`).focus();
    throw new Error("请填写忽略原因后再保存。");
  }
  const card = $(`#change-${id(change.id)}`);
  const controls = $$("button, textarea", card);
  controls.forEach((element) => {
    element.disabled = true;
  });
  try {
    await api(`/api/changes/${id(change.id)}`, {
      method: "PATCH",
      body: { status, note, revision: change.revision },
    });
    state.drafts.delete(change.id);
    state.comparison = await api(`/api/comparisons/${id(state.comparison.id)}`);
    const scrollY = window.scrollY;
    main.innerHTML = comparisonMarkup();
    window.scrollTo(0, scrollY);
    toast("审核意见已保存，操作记录已更新。");
    const focusTarget =
      $(`#note-${id(change.id)}`) || $('[data-action="filter-review"].active');
    if (focusTarget) focusTarget.focus({ preventScroll: true });
  } catch (error) {
    if (error.status === 409) {
      state.comparison = await api(
        `/api/comparisons/${id(state.comparison.id)}`,
      );
      main.innerHTML = comparisonMarkup();
      throw new Error(
        "此变更已被其他页面更新，已载入最新状态并保留你的未提交意见。请核对后重新保存。",
      );
    }
    throw error;
  } finally {
    controls.forEach((element) => {
      element.disabled = false;
    });
  }
}

STATUSES = {"pending": "待复核", "confirmed": "已确认", "ignored": "已忽略"}
KINDS = {"added": "新增章节", "removed": "删除章节", "modified": "内容修改"}
PRIORITIES = {"high": "重点关注", "normal": "常规"}


def _text(value: Any) -> str:
    if value is None:
        return ""
    if isinstance(value, (dict, list)):
        return json.dumps(value, ensure_ascii=False)
    return str(value)


def _html(value: Any) -> str:
    return escape(_text(value), quote=True)


def _csv_cell(value: Any) -> str:
    """Prevent spreadsheet applications interpreting untrusted cells as formulae.

    Leading whitespace can hide a formula marker, and tab/CR/LF themselves are
    unsafe leading characters in some spreadsheet importers. Prefix the entire
    original cell with a quote so the original content remains readable.
    """
    text = _text(value)
    trimmed = text.lstrip(" \t\r\n\v\f\ufeff")
    if (text[:1] in ("\t", "\r", "\n")) or (
        trimmed and trimmed[0] in "=+-@"
    ):
        return "'" + text
    return text


def _provenance(version: dict, label: str) -> str:
    fields = (
        ("版本名称", version.get("label")),
        ("版本日期", version.get("version_date")),
        ("资料来源", version.get("source")),
        ("原始文件", version.get("filename")),
        ("导入时间", version.get("created_at")),
        ("SHA-256", version.get("sha256")),
    )
    rows = "".join(
        f"<dt>{_html(key)}</dt><dd>{_html(value) or '—'}</dd>"
        for key, value in fields
    )
    warnings = version.get("warnings") or []
    warning_html = "".join(f"<li>{_html(w)}</li>" for w in warnings)
    if warning_html:
        warning_html = f'<ul class="warning">{warning_html}</ul>'
    return f'<section class="version"><h3>{_html(label)}</h3><dl>{rows}</dl>{warning_html}</section>'


def _change_html(change: dict, index: int) -> str:
    status = _text(change.get("status", "pending"))
    flags = change.get("flags") or []
    flag_html = "".join(f'<span class="flag">{_html(flag)}</span>' for flag in flags)
    segments = []
    for segment in change.get("segments") or []:
        text = _html(segment.get("text", ""))
        op = segment.get("op")
        if op == "delete":
            segments.append(f"<del>{text}</del>")
        elif op == "insert":
            segments.append(f"<ins>{text}</ins>")
        else:
            segments.append(text)
    inline = (
        '<details><summary>逐字差异（删除线为旧内容，下划线为新内容）</summary>'
        f'<pre class="inline-diff">{"".join(segments)}</pre></details>'
        if segments else ""
    )
    return f"""
    <article class="change">
      <header class="change-title"><h3>{index:02d} · {_html(change.get('section_title', '未命名章节'))}</h3>
      <span class="badge">{_html(KINDS.get(change.get('kind'), change.get('kind', '')))}</span>
      <span class="badge">{_html(PRIORITIES.get(change.get('priority'), change.get('priority', '')))}</span>
      <span class="badge">{_html(STATUSES.get(status, status))}</span></header>
      <div class="flags">{flag_html}</div>
      <div class="columns">
        <section><h4>旧版原文</h4><p class="location">{_html(change.get('old_location')) or '无对应章节'}</p>
          <pre>{_html(change.get('old_text')) or '（无）'}</pre></section>
        <section><h4>新版原文</h4><p class="location">{_html(change.get('new_location')) or '无对应章节'}</p>
          <pre>{_html(change.get('new_text')) or '（无）'}</pre></section>
      </div>
      {inline}
      <section class="review"><h4>人工审核记录</h4>
        <p>状态：{_html(STATUSES.get(status, status))}　·　操作员：{_html(change.get('reviewer')) or '—'}　·　记录修订号：{_html(change.get('revision', 0))}</p>

        <p class="note">{_html(change.get('note')) or '未填写审核意见'}</p>
      </section>
    </article>"""

def _csv_report(comparison: dict, exported_at: str) -> bytes:
    stream = io.StringIO(newline="")
    writer = csv.writer(stream)
    writer.writerow([
        "任务ID", "药品名称", "导出时间UTC", "旧版名称", "旧版日期", "旧版来源", "旧版文件", "旧版SHA256",
        "新版名称", "新版日期", "新版来源", "新版文件", "新版SHA256", "章节", "变化类型", "关注级别",
        "旧版位置", "新版位置", "旧版内容", "新版内容", "关注标记", "审核状态", "审核意见", "本地操作员（未认证）", "复核时间", "修订号",
    ])
    old, new = comparison.get("old_version") or {}, comparison.get("new_version") or {}
    common = [comparison.get("id"), comparison.get("drug_name"), exported_at]
    for version in (old, new):
        common.extend(version.get(key) for key in ("label", "version_date", "source", "filename", "sha256"))
    changes = comparison.get("changes") or []
    if not changes:
        writer.writerow([_csv_cell(value) for value in common + ["未检出章节内容差异"] + [""] * 12])
    for change in changes:
        row = common + [
            change.get("section_title"), KINDS.get(change.get("kind"), change.get("kind")),
            PRIORITIES.get(change.get("priority"), change.get("priority")),
            change.get("old_location"), change.get("new_location"), change.get("old_text"), change.get("new_text"),
            "；".join(_text(flag) for flag in change.get("flags") or []),
            STATUSES.get(change.get("status"), change.get("status")), change.get("note"),
            change.get("reviewer"), change.get("reviewed_at"), change.get("revision"),
        ]
        writer.writerow([_csv_cell(value) for value in row])
    return stream.getvalue().encode("utf-8-sig")


def export_report(comparison: dict, format: str) -> tuple[bytes, str, str]:
    """Return a ready-to-download body, content type, and safe ASCII filename."""
    exported_at = datetime.now(timezone.utc).isoformat(timespec="seconds")
    identifier = str(comparison.get("id", "report"))
    safe_id = "".join(char for char in identifier if char.isascii() and (char.isalnum() or char in "-_")) or "report"
    if format == "html":
        return _html_report(comparison, exported_at), "text/html; charset=utf-8", f"comparison-{safe_id}.html"
    if format == "csv":
        return _csv_report(comparison, exported_at), "text/csv; charset=utf-8", f"comparison-{safe_id}.csv"
    if format == "json":
        payload = dict(comparison)
        payload["exported_at"] = exported_at
        return json.dumps(payload, ensure_ascii=False, indent=2).encode("utf-8"), "application/json; charset=utf-8", f"comparison-{safe_id}.json"
    raise ValueError("不支持的导出格式，请选择 html、csv 或 json。")
